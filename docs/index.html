Terminal extract.py 
from pydantic import BaseModel
from openextract import extract

class Report(BaseModel):
    title: str
    findings: list[str]
    severity: int

result = extract(
    schema=Report,
    model="openai:gpt-5.2",
    input_file="https://example.com/report.pdf",
    instructions="Extract findings",
)
